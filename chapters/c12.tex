\section{Максимальная функция Харди-Литтлвуда и дифференцируемость интеграла Лебега}
\begin{definition}
    $\widetilde{L}_1^{loc}(\R^n, \mathcal{L}^n)$ — множество всех измеримых в широком смысле на $\R^n$ функций, которые интегрируемы по Лебегу на любом шаре в $\R^n$.
\end{definition} 
\begin{definition}
    Пусть $f \in \widetilde{L}_1^{loc}(\R^n, \mathcal{L}^n)$, то $$M_{[f]}(x) := \sup\limits_{r>0} \fint_{B_r(x)}|f(y)|d \mathcal{L}^n(y) = \frac{1}{\mathcal{L}^n(B_r(x))} \int_{B_r(x)} |f(y)|d \mu(y),$$
    $M_{[f]}(x) \in [0, +\infty]$.
\end{definition}
